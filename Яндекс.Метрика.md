##### Как можно реализовать
###### Параметры сбора (настроить вывод в файле Stats.js): общая статистика, посещения, уникальные посетители, просмотры страниц, процент отказов, среднее время визита, глубина просмотра, популярные страницы, устройства, источники трафика
```python
YANDEX_TOKEN = os.getenv("YANDEX_METRIKA_TOKEN")
COUNTER_ID = os.getenv("YANDEX_METRIKA_COUNTER_ID")
BASE_URL = "https://api-metrika.yandex.net/stat/v1/data"

def get_visits_and_views():
    headers = { "Authorization": f"OAuth {YANDEX_TOKEN}" }
    params = {
        "ids": COUNTER_ID,
        # основные метрики
        "metrics": (
            "ym:s:visits,"
            "ym:s:users,"
            "ym:s:pageviews,"
            "ym:s:bounceRate,"
            "ym:s:avgVisitDurationSeconds,"
            "ym:s:pageDepth"
        ),
        "dimensions": "ym:s:date",
        "date1": "30daysAgo",
        "date2": "today",
        "sort": "ym:s:date"
    }
    response = requests.get(BASE_URL, headers=headers, params=params)
    response.raise_for_status()
    data = response.json()

    # популярные страницы
    pages_params = {
        "ids": COUNTER_ID,
        "metrics": "ym:pv:pageviews",
        "dimensions": "ym:pv:URLPath",
        "date1": "30daysAgo",
        "date2": "today",
        "sort": "-ym:pv:pageviews",
        "limit": 10
    }
    pages_response = requests.get(BASE_URL, headers=headers, params=pages_params)
    pages_response.raise_for_status()

    # устройства
    devices_params = {
        "ids": COUNTER_ID,
        "metrics": "ym:s:visits",
        "dimensions": "ym:s:deviceCategory",
        "date1": "30daysAgo",
        "date2": "today"
    }
    devices_response = requests.get(BASE_URL, headers=headers, params=devices_params)
    devices_response.raise_for_status()

    # источники трафика
    traffic_params = {
        "ids": COUNTER_ID,
        "metrics": "ym:s:visits",
        "dimensions": "ym:s:lastTrafficSource",
        "date1": "30daysAgo",
        "date2": "today"
    }
    traffic_response = requests.get(BASE_URL, headers=headers, params=traffic_params)
    traffic_response.raise_for_status()

    return {
        "overview": data,
        "popular_pages": pages_response.json(),
        "devices": devices_response.json(),
        "traffic_sources": traffic_response.json()
    }
```
